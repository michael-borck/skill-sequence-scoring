Sequence Sensitivity Explorer

Same frames, different order — see how the score changes

Build a 9-frame sequence by clicking the buttons below. Frame 10 is fixed as an open frame (5, 4) to isolate the effect of frame ordering. Both scores update live.

function scoreTraditional(balls) {
  let score = 0;
  let i = 0;
  for (let frame = 0; frame < 10; frame++) {
    if (i >= balls.length) return null;
    if (frame < 9) {
      if (balls[i] === 10) {
        if (i + 2 >= balls.length) return null;
        score += 10 + balls[i+1] + balls[i+2];
        i += 1;
      } else {
        if (i + 1 >= balls.length) return null;
        score += balls[i] + balls[i+1];
        if (balls[i] + balls[i+1] === 10) {
          if (i + 2 >= balls.length) return null;
          score += balls[i+2];
        }
        i += 2;
      }
    } else {
      if (balls[i] === 10) {
        if (i + 2 >= balls.length) return null;
        score += 10 + balls[i+1] + balls[i+2];
      } else {
        if (i + 1 >= balls.length) return null;
        if (balls[i] + balls[i+1] === 10) {
          if (i + 2 >= balls.length) return null;
          score += 10 + balls[i+2];
        } else {
          score += balls[i] + balls[i+1];
        }
      }
    }
  }
  return score;
}

function scoreWorld(balls) {
  let score = 0;
  let i = 0;
  for (let frame = 0; frame < 10; frame++) {
    if (i >= balls.length) return null;
    if (balls[i] === 10) {
      score += 30;
      i += 1;
    } else {
      if (i + 1 >= balls.length) return null;
      if (balls[i] + balls[i+1] === 10) {
        score += 10 + balls[i];
      } else {
        score += balls[i] + balls[i+1];
      }
      i += 2;
    }
  }
  return score;
}

How to use

  1. Click frame buttons to build a 9-frame sequence
  2. Watch both scores update as you reach 9 frames
  3. Try reordering the same frames — click Clear All and enter them in a different order
  4. Notice: World Bowling always gives the same score for the same frame composition, regardless of order. Traditional scoring changes.

Try these experiments

Experiment 1: Strike placement matters

  • Enter: X, X, X, X, X, 5/, 5/, 5/, 5/ — note both scores
  • Clear and enter: 5/, 5/, 5/, 5/, X, X, X, X, X — note the traditional score changes, World Bowling stays the same

Experiment 2: Consecutive vs scattered strikes

  • Enter: X, X, X, 5,4, 5,4, 5,4, 5,4, 5,4, 5,4
  • Clear and enter: X, 5,4, X, 5,4, X, 5,4, 5,4, 5,4, 5,4
  • Same three strikes, but consecutive strikes score higher under traditional scoring
