How much is each consecutive strike worth?

What is “marginal value”?

If you replace one open frame with a strike, your total game score goes up. The marginal value is how much it goes up. For example, with a (5,4) open frame:

  • The open scores 5 + 4 = 9 points
  • A strike under World Bowling scores 30 points
  • So the marginal value is 30 − 9 = +21 points — the same every time, regardless of when you throw it

Under traditional scoring, the marginal value depends on context. A lone strike earns only +10 (because the bonus depends on the next two balls, which are just an open frame). But a strike that extends a run of consecutive strikes earns much more — up to +37 for the 10th strike in a perfect game — because it compounds the bonuses of preceding strikes.

Use the slider below to see this in action. Try changing the fill type to confirm the pattern holds regardless of what the non-strike frames score.

function scoreTraditional(balls) {
  let score = 0, i = 0;
  for (let frame = 0; frame < 10; frame++) {
    if (i >= balls.length) return null;
    if (frame < 9) {
      if (balls[i] === 10) {
        if (i + 2 >= balls.length) return null;
        score += 10 + balls[i+1] + balls[i+2];
        i += 1;
      } else {
        if (i + 1 >= balls.length) return null;
        score += balls[i] + balls[i+1];
        if (balls[i] + balls[i+1] === 10) {
          if (i + 2 >= balls.length) return null;
          score += balls[i+2];
        }
        i += 2;
      }
    } else {
      if (balls[i] === 10) {
        if (i + 2 >= balls.length) return null;
        score += 10 + balls[i+1] + balls[i+2];
      } else {
        if (i + 1 >= balls.length) return null;
        if (balls[i] + balls[i+1] === 10) {
          if (i + 2 >= balls.length) return null;
          score += 10 + balls[i+2];
        } else {
          score += balls[i] + balls[i+1];
        }
      }
    }
  }
  return score;
}

function scoreWorld(balls) {
  let score = 0, i = 0;
  for (let frame = 0; frame < 10; frame++) {
    if (i >= balls.length) return null;
    if (balls[i] === 10) {
      score += 30; i += 1;
    } else {
      if (i + 1 >= balls.length) return null;
      if (balls[i] + balls[i+1] === 10) score += 10 + balls[i];
      else score += balls[i] + balls[i+1];
      i += 2;
    }
  }
  return score;
}

gradientData = {
  let rows = [];
  for (let n = 0; n <= 10; n++) {
    let balls = [];
    for (let i = 0; i < n; i++) balls.push(10);
    let remaining = 10 - n;
    for (let i = 0; i < remaining; i++) balls = balls.concat(fillBalls);
    if (n === 10) {
      balls = [10,10,10,10,10,10,10,10,10,10,10,10];
    } else if (n === 9) {
      balls.pop(); balls.pop();
      balls.push(10);
      balls = balls.concat(fillBalls);
    }
    let t = scoreTraditional(balls);
    let w = scoreWorld(balls);
    let tm = n > 0 ? t - rows[n-1].trad : 0;
    let wm = n > 0 ? w - rows[n-1].world : 0;
    rows.push({strikes: n, trad: t, world: w, tradMarginal: tm, worldMarginal: wm, current: n === nStrikes});
  }
  return rows;
}

What to notice

  • Traditional marginal value increases as the run extends — the 10th strike is worth far more than the 1st
  • World Bowling marginal is flat — every strike adds the same amount regardless of context
  • Change the fill type to see that this pattern holds regardless of what the non-strike frames score
  • The compounding effect is a structural property of the bonus system, not an artefact of any particular example
